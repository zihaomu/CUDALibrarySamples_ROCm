#!/usr/bin/env python3
"""Report raw timings while refusing a speed ratio under GPU contention."""
import json
from catalog import ROOT
p=ROOT/'migration/reports';a=json.loads((p/'benchmark-rocm.json').read_text());b=json.loads((p/'benchmark-cuda.json').read_text());a['program_status']='PASSED' if a['return_code']==0 else 'FAILED';a['status']='BLOCKED';a['performance_qualified']=False;a['reason']='GPU occupied by unrelated workloads; idle/memory gate failed. Raw observations are not qualified performance comparisons.';(p/'benchmark-rocm.json').write_text(json.dumps(a,indent=2)+'\n');rows=[]
for ra in a['profiles']:
 rb=next(v for v in b['profiles'] if (v['dtype'],v['n'])==(ra['dtype'],ra['n']));rows.append({'dtype':ra['dtype'],'n':ra['n'],'rocm_raw_median_ms':ra['gpu_median_ms'],'cuda_raw_median_ms':rb['gpu_median_ms'],'qualified_speed_ratio':None,'status':'BLOCKED','rocm_groups_ms':ra['group_medians_ms'],'cuda_groups_ms':rb['group_medians_ms'],'correctness_reference_error':{'rocm':ra['max_reference_error'],'cuda':rb['max_reference_error']}})
r={'status':'BLOCKED','reason':a['reason'],'scope':'6 GEMM profiles only, FP32/FP64 n256/1024/2048. GPU event timing around GEMM, allocation/transfer/init excluded, >=1s warmup then 3x100 iterations. Independent sampled long-double CPU reference.','rocm_evidence':a['evidence'],'cuda_evidence':b['evidence'],'qualification_evidence':['migration/reports/performance-idle-gate.log','migration/environments/benchmark-amd-final-before.json','migration/environments/benchmark-amd-final-after.json'],'retry_condition':'An allowed unreserved AMD GPU must satisfy >=90% free memory and <=2% busy, then remain uncontended throughout both matched runs; retain power/frequency/topology snapshots.','profiles':rows};(p/'performance-comparison.json').write_text(json.dumps(r,indent=2)+'\n')
text='# GEMM 性能实验结果\n\n**BLOCKED：AMD GPU 有其它任务，未满足合格性能对比条件。** 下列仅为竞争条件下的原始观察值，不计算速度比或移植加速收益。正确性与性能资格分别记录。\n\n'+r['scope']+'\n\n| 精度 | n | AMD 原始 median ms（受竞争） | CUDA 原始 median ms | 合格速度比 |\n| --- | ---: | ---: | ---: | --- |\n'
for v in rows:text+=f'| {v["dtype"]} | {v["n"]} | {v["rocm_raw_median_ms"]:.6f} | {v["cuda_raw_median_ms"]:.6f} | 不报告 |\n'
text+='\n空闲资格失败日志：[performance-idle-gate.log](performance-idle-gate.log)。完整三组 median、参考误差、运行目录和恢复条件：[performance-comparison.json](performance-comparison.json)。\n';(p/'performance-comparison.md').write_text(text);print('Performance:',r['status'],len(rows),'profiles')
