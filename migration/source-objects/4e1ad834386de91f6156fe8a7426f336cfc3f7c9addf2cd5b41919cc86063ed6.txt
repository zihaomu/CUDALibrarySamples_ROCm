// Project 2:4 CSR-free packed format; not the opaque cuSPARSELt binary layout.
#pragma once
#include "rewrite_runtime.hpp"
#include <limits>
namespace rewrite_sparse24 {
using namespace rewrite;
struct Shape{int m,n,k,batch,lda,ldb,ldc,sa,sb,sc,bt,advanced,search;float alpha,beta,threshold,upper;};
inline int integer(const Config&c,const char*key,int low,int high){double x=c.get(key,-1);if(!std::isfinite(x)||std::floor(x)!=x||x<low||x>high)throw std::invalid_argument(std::string("descriptor ")+key);return int(x);}
inline float scalar(const Config&c,const char*key){auto it=c.values.find(key);if(it==c.values.end()||!std::isfinite(it->second)||!std::isfinite(float(it->second)))throw std::invalid_argument(std::string("scalar ")+key);return float(it->second);}
inline Shape shape(const Config&c,bool advanced){Shape s;integer(c,"dtype",16,16);s.m=integer(c,"m",4,128);s.n=integer(c,"n",1,128);s.k=integer(c,"k",4,256);s.batch=integer(c,"batch",1,4);s.lda=integer(c,"lda",s.k,s.k+128);s.bt=integer(c,"transpose_b",0,1);s.ldb=integer(c,"ldb",s.bt?s.k:s.n,384);s.ldc=integer(c,"ldc",s.n,s.n+128);s.sa=integer(c,"stride_a",s.m*s.lda,1000000);s.sb=integer(c,"stride_b",(s.bt?s.n:s.k)*s.ldb,1000000);s.sc=integer(c,"stride_c",s.m*s.ldc,1000000);s.search=integer(c,"search",0,1);s.advanced=advanced;s.alpha=scalar(c,"alpha");s.beta=scalar(c,"beta");s.threshold=scalar(c,"relu_threshold");s.upper=scalar(c,"relu_upper");if(s.m%4||s.k%4||!std::isfinite(s.alpha)||!std::isfinite(s.beta)||!std::isfinite(s.threshold)||!std::isfinite(s.upper)||s.upper<s.threshold)throw std::invalid_argument("tile or scalar descriptor");return s;}
inline std::vector<uint16_t> legal_tiles(){std::vector<uint16_t>masks;for(unsigned v=0;v<65536;++v){bool valid=true;for(int r=0;r<4;++r){int row=0,column=0;for(int c=0;c<4;++c){row+=(v>>(4*r+c))&1;column+=(v>>(4*c+r))&1;}valid=valid&&row==2&&column==2;}if(valid)masks.push_back(v);}if(masks.size()!=90)throw std::runtime_error("tile-mask census");return masks;}
__global__ void prune_tile(Shape s,const __half*input,__half*out,const uint16_t*legal,uint16_t*selected){int t=blockIdx.x*blockDim.x+threadIdx.x,tiles=(s.m/4)*(s.k/4);if(t>=s.batch*tiles)return;int batch=t/tiles,local=t%tiles,row=4*(local/(s.k/4)),col=4*(local%(s.k/4));double best=-1;uint16_t chosen=0;for(int candidate=0;candidate<90;++candidate){uint16_t mask=legal[candidate];double score=0;for(int r=0;r<4;++r)for(int c=0;c<4;++c)if(mask&(1u<<(4*r+c))){double value=__half2float(input[batch*s.sa+(row+r)*s.lda+col+c]);score+=value<0?-value:value;}if(score>best){best=score;chosen=mask;}}selected[t]=chosen;for(int r=0;r<4;++r)for(int c=0;c<4;++c)out[batch*s.sa+(row+r)*s.lda+col+c]=(chosen&(1u<<(4*r+c)))?input[batch*s.sa+(row+r)*s.lda+col+c]:__float2half(0);}
__global__ void compress24(Shape s,const __half*pruned,const uint16_t*tiles,__half*values,uint8_t*metadata){int t=blockIdx.x*blockDim.x+threadIdx.x,groups=s.m*(s.k/4);if(t>=s.batch*groups)return;int batch=t/groups,local=t%groups,row=local/(s.k/4),g=local%(s.k/4);uint16_t mask=tiles[batch*(s.m/4)*(s.k/4)+(row/4)*(s.k/4)+g];int selected[2],count=0;for(int c=0;c<4;++c)if(mask&(1u<<(4*(row%4)+c)))selected[count++]=c;if(count!=2)return;metadata[t]=uint8_t(selected[0]|(selected[1]<<2));values[2*t]=pruned[batch*s.sa+row*s.lda+4*g+selected[0]];values[2*t+1]=pruned[batch*s.sa+row*s.lda+4*g+selected[1]];}
__global__ void check24(Shape s,const uint8_t*metadata,int*status){int t=blockIdx.x*blockDim.x+threadIdx.x,tiles=(s.m/4)*(s.k/4);if(t>=s.batch*tiles)return;int batch=t/tiles,local=t%tiles,row=4*(local/(s.k/4)),g=local%(s.k/4);int columns[4]={};for(int r=0;r<4;++r){uint8_t code=metadata[batch*s.m*(s.k/4)+(row+r)*(s.k/4)+g];int a=code&3,b=(code>>2)&3;if(code>=16||a>=b)atomicExch(status,1);columns[a]++;columns[b]++;}for(int c=0;c<4;++c)if(columns[c]!=2)atomicExch(status,1);}
template<int Mapping> __global__ void sparse_matmul24(Shape s,const __half*values,const uint8_t*metadata,const __half*b,const __half*c,const __half*bias,__half*d){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t>=s.batch*s.m*s.n)return;int batch=t/(s.m*s.n),local=t%(s.m*s.n),row=Mapping?local%s.m:local/s.n,column=Mapping?local/s.m:local%s.n;float sum=0;for(int g=0;g<s.k/4;++g){int group=batch*s.m*(s.k/4)+row*(s.k/4)+g;uint8_t code=metadata[group];int k0=4*g+(code&3),k1=4*g+((code>>2)&3);int b0=batch*s.sb+(s.bt?column*s.ldb+k0:k0*s.ldb+column),b1=batch*s.sb+(s.bt?column*s.ldb+k1:k1*s.ldb+column);sum+=__half2float(values[2*group])*__half2float(b[b0]);sum+=__half2float(values[2*group+1])*__half2float(b[b1]);}float value=s.alpha*sum+s.beta*__half2float(c[batch*s.sc+row*s.ldc+column]);if(s.advanced){value+=__half2float(bias[row]);value=value<=s.threshold?0:(value<s.upper?value:s.upper);}d[batch*s.sc+row*s.ldc+column]=__float2half_rn(value);}
inline int run(bool advanced){try{Config config;auto s=shape(config,advanced);identity();auto a=read<__half>("A",s.batch*s.sa),b=read<__half>("B",s.batch*s.sb),c=read<__half>("C",s.batch*s.sc),bias=read<__half>("bias",s.m);for(auto*input:{&a,&b,&c,&bias})for(__half value:*input)if(!std::isfinite(__half2float(value)))throw std::invalid_argument("finite FP16 inputs required");Buffer<__half>da(a),ap(a),db(b),dc(c),dd(c),di(bias);auto masks=legal_tiles();Buffer<uint16_t>legal(masks),selected(s.batch*(s.m/4)*(s.k/4));int groups=s.batch*s.m*(s.k/4);Buffer<__half>packed(2*groups);Buffer<uint8_t>metadata(groups);Buffer<int>status(std::vector<int>{0});int tiles=s.batch*(s.m/4)*(s.k/4);prune_tile<<<(tiles+127)/128,128>>>(s,da.p,ap.p,legal.p,selected.p);compress24<<<(groups+255)/256,256>>>(s,ap.p,selected.p,packed.p,metadata.p);check24<<<(tiles+127)/128,128>>>(s,metadata.p,status.p);sync();if(status.get()[0])throw std::runtime_error("GPU prune/compress check");
 auto launch=[&](int algorithm){int outputs=s.batch*s.m*s.n;if(algorithm==0)sparse_matmul24<0><<<(outputs+255)/256,256>>>(s,packed.p,metadata.p,db.p,dc.p,di.p,dd.p);else sparse_matmul24<1><<<(outputs+255)/256,256>>>(s,packed.p,metadata.p,db.p,dc.p,di.p,dd.p);};int chosen=0;std::vector<float>timings;std::vector<int>candidates;
 if(s.search){for(int algorithm=0;algorithm<2;++algorithm){launch(algorithm);sync();hipEvent_t start,stop;RW_CHECK(hipEventCreate(&start));RW_CHECK(hipEventCreate(&stop));RW_CHECK(hipEventRecord(start));for(int trial=0;trial<3;++trial)launch(algorithm);RW_CHECK(hipEventRecord(stop));RW_CHECK(hipEventSynchronize(stop));float ms=0;RW_CHECK(hipEventElapsedTime(&ms,start,stop));RW_CHECK(hipEventDestroy(start));RW_CHECK(hipEventDestroy(stop));timings.push_back(ms/3);candidates.push_back(algorithm);}chosen=timings[1]<timings[0]?1:0;}
 launch(chosen);sync();emit("A_pruned",ap.get());emit("tile_masks",selected.get());emit("packed_values",packed.get());emit("metadata",metadata.get());emit("D",dd.get());emit("A_after",da.get());emit("B_after",db.get());emit("C_after",dc.get());emit("bias_after",di.get());emit("prune_check",status.get());emit("search_candidates",candidates);emit("search_ms",timings);emit("plan",std::vector<int64_t>{1,0,0,int64_t(2*groups*sizeof(__half)),int64_t(groups),s.k/2,chosen,s.search});std::cout<<"project sparse 2:4: "<<2*groups<<" packed values; "<<groups<<" metadata bytes; selected algorithm "<<chosen<<'\n';return 0;}catch(const std::exception&e){std::cerr<<e.what()<<'\n';return 1;}}
}
