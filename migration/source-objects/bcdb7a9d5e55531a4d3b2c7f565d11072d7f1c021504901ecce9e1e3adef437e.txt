#include <cstdio>
#include <cstdlib>
#include <vector>

#include <hipblas.h>
#include <hip/hip_runtime.h>

#include "cublas_utils.h"

using data_type = double;

int main(int argc, char *argv[]) {
// MIGRATION_BATCH2: default is unchanged original input and mathematical GPU API.
const std::string migration_case=std::getenv("MIGRATION_CASE")?std::getenv("MIGRATION_CASE"):"original";

    hipblasHandle_t cublasH = NULL;
    hipStream_t stream = NULL;

    /*
     *   A = 1.0
     *   B = 5.0
     *   X = 2.1
     *   Y = 1.2
     */

    data_type A = 1.0;
    data_type B = 5.0;
    data_type X = 2.1;
    data_type Y = 1.2;
    std::vector<data_type> param = {1.0, 5.0, 6.0, 7.0, 8.0}; // flag = param[0]

    data_type *d_A = nullptr;
    data_type *d_B = nullptr;

    printf("A\n");
    std::printf("%0.2f\n", A);
    printf("=====\n");

    printf("B\n");
    std::printf("%0.2f\n", B);
    printf("=====\n");

    printf("X\n");
    std::printf("%0.2f\n", X);
    printf("=====\n");

    printf("Y\n");
    std::printf("%0.2f\n", Y);
    printf("=====\n");

    printf("param\n");
    print_vector(param.size(), param.data());
    printf("=====\n");

if(migration_case=="zero"){Y=0;}if(migration_case=="negative"){A=-1;}if(migration_case=="rescale"){A=1e-20;B=2e-20;X=2;Y=1;}
migration_observer::emit("input_A",&A,1);
migration_observer::emit("input_B",&B,1);
migration_observer::emit("input_X",&X,1);
migration_observer::emit("input_Y",&Y,1);
migration_observer::emit("input_param",param.data(),param.size());
    /* step 1: create hipblas handle, bind a stream */
    CUBLAS_CHECK(hipblasCreate(&cublasH));

    CUDA_CHECK(hipStreamCreateWithFlags(&stream, hipStreamNonBlocking));
    CUBLAS_CHECK(hipblasSetStream(cublasH, stream));

    /* step 3: compute */
    CUBLAS_CHECK(hipblasDrotmg(cublasH, &A, &B, &X, &Y, param.data()));

    CUDA_CHECK(hipStreamSynchronize(stream));

    /*
     *   A = 3.10
     *   B = 0.62
     *   X = 1.94
     */

    printf("A\n");
    std::printf("%0.2f\n", A);
    printf("=====\n");

    printf("B\n");
    std::printf("%0.2f\n", B);
    printf("=====\n");

    printf("X\n");
    std::printf("%0.2f\n", X);
    printf("=====\n");

    
// MIGRATION_OBSERVE_ORIGINAL: host buffers after GPU synchronization.
migration_observer::emit("param",param.data(),param.size());

// MIGRATION_OBSERVE_SCALARS
migration_observer::emit("A",&A,1);
migration_observer::emit("B",&B,1);
migration_observer::emit("X",&X,1);
migration_observer::emit("Y",&Y,1);
migration_observer::emit("A",&A,1);
migration_observer::emit("B",&B,1);
migration_observer::emit("X",&X,1);
migration_observer::emit("Y",&Y,1);
migration_observer::emit("param",param.data(),param.size());
/* free resources */
    CUDA_CHECK(hipFree(d_A));
    CUDA_CHECK(hipFree(d_B));

    CUBLAS_CHECK(hipblasDestroy(cublasH));

    CUDA_CHECK(hipStreamDestroy(stream));

    CUDA_CHECK(hipDeviceReset());

    return EXIT_SUCCESS;
}
