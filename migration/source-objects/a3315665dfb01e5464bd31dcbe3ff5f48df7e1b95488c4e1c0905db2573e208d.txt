#include <cstdio>
#include <cstdlib>
#include <vector>

#include <cuda_runtime.h>
#include <cusolverDn.h>

#include "cusolver_utils.h"

int main(int argc, char *argv[]) {
// MIGRATION_BATCH3: preserve original GPU mathematical calls.
const std::string migration_case=std::getenv("MIGRATION_CASE")?std::getenv("MIGRATION_CASE"):"original";

    cusolverDnHandle_t cusolverH = NULL;
    cudaStream_t stream = NULL;

    const int m = 3;
    const int lda = m;
    /*
     *       | 3.5 0.5 0.0 |
     *   A = | 0.5 3.5 0.0 |
     *       | 0.0 0.0 2.0 |
     *
     */
    const std::vector<double> A = {3.5, 0.5, 0.0, 0.5, 3.5, 0.0, 0.0, 0.0, 2.0};
    const std::vector<double> lambda = {2.0, 3.0, 4.0};

    std::vector<double> V(lda * m, 0); // eigenvectors
    std::vector<double> W(m, 0);       // eigenvalues

    double *d_A = nullptr;
    double *d_W = nullptr;
    int *d_info = nullptr;

    int info = 0;

    double vl = 0.0;
    double vu = 0.0;
    int il = 1;
    int iu = 2;
    int h_meig = 0;

    int lwork = 0;            /* size of workspace */
    double *d_work = nullptr; /* device workspace*/

    std::printf("A = (matlab base-1)\n");
    print_matrix(m, m, A.data(), lda);
    std::printf("=====\n");

    /* step 1: create cusolver handle, bind a stream */
    CUSOLVER_CHECK(cusolverDnCreate(&cusolverH));

    CUDA_CHECK(cudaStreamCreateWithFlags(&stream, cudaStreamNonBlocking));
    CUSOLVER_CHECK(cusolverDnSetStream(cusolverH, stream));

    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_A), sizeof(double) * A.size()));
    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_W), sizeof(double) * W.size()));
    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_info), sizeof(int)));

    CUDA_CHECK(
        cudaMemcpyAsync(d_A, A.data(), sizeof(double) * A.size(), cudaMemcpyHostToDevice, stream));

    // step 3: query working space of syevd
    cusolverEigMode_t jobz = CUSOLVER_EIG_MODE_VECTOR; // compute eigenvalues and eigenvectors.
    cusolverEigRange_t range = CUSOLVER_EIG_RANGE_I;
if(migration_case=="value-range"){range=CUSOLVER_EIG_RANGE_V;vl=2.5;vu=4.5;}
if(migration_case=="all-range")range=CUSOLVER_EIG_RANGE_ALL;
if(migration_case=="empty-range"){range=CUSOLVER_EIG_RANGE_V;vl=10.;vu=11.;}
   // eigenvalues/eigenvectors in the half-open
                                                       // interval (vl,vu] will be found
    cublasFillMode_t uplo = CUBLAS_FILL_MODE_LOWER;

    CUSOLVER_CHECK(cusolverDnDsyevdx_bufferSize(cusolverH, jobz, range, uplo, m, d_A, lda, vl, vu,
                                                il, iu, &h_meig, d_W, &lwork));

    CUDA_CHECK(cudaMalloc(reinterpret_cast<void **>(&d_work), sizeof(double) * lwork));

    // step 4: compute spectrum
    CUSOLVER_CHECK(cusolverDnDsyevdx(cusolverH, jobz, range, uplo, m, d_A, lda, vl, vu, il, iu,
                                     &h_meig, d_W, d_work, lwork, d_info));

    CUDA_CHECK(
        cudaMemcpyAsync(V.data(), d_A, sizeof(double) * V.size(), cudaMemcpyDeviceToHost, stream));
    CUDA_CHECK(
        cudaMemcpyAsync(W.data(), d_W, sizeof(double) * W.size(), cudaMemcpyDeviceToHost, stream));
    CUDA_CHECK(cudaMemcpyAsync(&info, d_info, sizeof(int), cudaMemcpyDeviceToHost, stream));

    CUDA_CHECK(cudaStreamSynchronize(stream));

    std::printf("after syevd: info = %d\n", info);
    if (0 > info) {
        std::printf("%d-th parameter is wrong \n", -info);
        exit(1);
    }

    std::printf("Eigenvalues found = %d\n",h_meig);
    for(int i=0;i<h_meig;++i)std::printf("W[%d] = %.17g\n",i,W[i]);
    print_matrix(m,h_meig,V.data(),lda);
// MIGRATION_OBSERVE_ORIGINAL: host buffers after GPU synchronization.
migration_observer::emit("A",A.data(),A.size());
migration_observer::emit("lambda",lambda.data(),lambda.size());
migration_observer::emit("V",V.data(),m*h_meig);
migration_observer::emit("W",W.data(),h_meig);
migration_observer::emit("info",&info,1);
migration_observer::emit("meig",&h_meig,1);
migration_observer::emit("input_il",&il,1);
migration_observer::emit("input_iu",&iu,1);
migration_observer::emit("input_vl",&vl,1);
migration_observer::emit("input_vu",&vu,1);
migration_observer::emit("input_A",A.data(),A.size());
/* free resources */
    CUDA_CHECK(cudaFree(d_A));
    CUDA_CHECK(cudaFree(d_W));
    CUDA_CHECK(cudaFree(d_info));
    CUDA_CHECK(cudaFree(d_work));

    CUSOLVER_CHECK(cusolverDnDestroy(cusolverH));

    CUDA_CHECK(cudaStreamDestroy(stream));

    CUDA_CHECK(cudaDeviceReset());

    return EXIT_SUCCESS;
}
