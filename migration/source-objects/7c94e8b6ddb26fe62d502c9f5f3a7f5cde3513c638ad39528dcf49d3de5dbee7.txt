// Full FP32 C_munv=alpha*A_mhkn*B_ukvh+beta*C_munv using HIP permutations + rocBLAS.
#include "check.hpp"
#include <rocblas/rocblas.h>
using namespace sample;
struct Dims{size_t M,N,U,V,H,K;};
__global__ void packA(const float*in,float*out,Dims d){size_t i=blockIdx.x*blockDim.x+threadIdx.x,total=d.M*d.N*d.H*d.K;if(i>=total)return;size_t r=i%(d.M*d.N),q=i/(d.M*d.N),m=r%d.M,n=r/d.M,h=q%d.H,k=q/d.H;out[i]=in[m+d.M*(h+d.H*(k+d.K*n))];}
__global__ void packB(const float*in,float*out,Dims d){size_t i=blockIdx.x*blockDim.x+threadIdx.x,total=d.U*d.V*d.H*d.K;if(i>=total)return;size_t q=i%(d.H*d.K),c=i/(d.H*d.K),h=q%d.H,k=q/d.H,u=c%d.U,v=c/d.U;out[i]=in[u+d.U*(k+d.K*(v+d.V*h))];}
__global__ void permC(const float*in,float*out,Dims d,bool unpack){size_t i=blockIdx.x*blockDim.x+threadIdx.x,total=d.M*d.N*d.U*d.V;if(i>=total)return;size_t r=i%(d.M*d.N),c=i/(d.M*d.N),m=r%d.M,n=r/d.M,u=c%d.U,v=c/d.U,raw=m+d.M*(u+d.U*(n+d.N*v));if(unpack)out[raw]=in[i];else out[i]=in[raw];}
void test(Dims d,float beta,const char*name){size_t na=d.M*d.N*d.H*d.K,nb=d.U*d.V*d.H*d.K,nc=d.M*d.N*d.U*d.V;std::vector<float>a(na),b(nb),c(nc);for(size_t i=0;i<na;++i)a[i]=(int(i%19)-9)*.125f;for(size_t i=0;i<nb;++i)b[i]=(int(i%17)-8)*.125f;for(size_t i=0;i<nc;++i)c[i]=(int(i%13)-6)*.125f;Device<float>A(a),B(b),C(c),PA(na),PB(nb),PC(nc);packA<<<(na+255)/256,256>>>(A.p,PA.p,d);packB<<<(nb+255)/256,256>>>(B.p,PB.p,d);permC<<<(nc+255)/256,256>>>(C.p,PC.p,d,false);CHECK(hipGetLastError());rocblas_handle h;CHECK(rocblas_create_handle(&h));float alpha=1.1f;CHECK(rocblas_sgemm(h,rocblas_operation_none,rocblas_operation_none,d.M*d.N,d.U*d.V,d.H*d.K,&alpha,PA.p,d.M*d.N,PB.p,d.H*d.K,&beta,PC.p,d.M*d.N));permC<<<(nc+255)/256,256>>>(PC.p,C.p,d,true);CHECK(hipGetLastError());CHECK(hipDeviceSynchronize());auto output=C.get();for(auto z:output)if(!std::isfinite(z))throw std::runtime_error("nonfinite tensor output");size_t checks=nc<1024?nc:32;std::vector<float>actual(checks),expected(checks);for(size_t p=0;p<checks;++p){size_t index=checks==nc?p:(p*130787+17)%nc,q=index,m=q%d.M;q/=d.M;size_t u=q%d.U;q/=d.U;size_t n=q%d.N,v=q/d.N;double sum=0;for(size_t k=0;k<d.K;++k)for(size_t hh=0;hh<d.H;++hh)sum+=double(a[m+d.M*(hh+d.H*(k+d.K*n))])*b[u+d.U*(k+d.K*(v+d.V*hh))];actual[p]=output[index];expected[p]=alpha*sum+beta*c[index];}report(name,actual,expected,2e-4,2e-5);CHECK(rocblas_destroy_handle(h));}
int main(){return run([]{test({4,3,2,5,2,7},.5f,"tensor4d-small-full-cpu-beta");test({96,96,96,64,64,64},0.f,"tensor4d-original-extents-fp32-cpu32-samples");});}
