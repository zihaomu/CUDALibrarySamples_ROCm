__device__ float sqrt15(float x){
#ifdef __HIP_PLATFORM_AMD__
 return __builtin_sqrtf(x);
#else
 return sqrtf(x);
#endif
}
__device__ double root15(double x){
#ifdef __HIP_PLATFORM_AMD__
 return __builtin_sqrt(x);
#else
 return sqrt(x);
#endif
}
__device__ int clamp15(int v,int hi){return v<0?0:v>=hi?hi-1:v;}
__global__ void gradient15(const unsigned char*src,int stride,int rx,int ry,int w,int h,int channels,int*gx,int*gy,float*mag){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){int x=t%w,y=t/w,best=-1,dx=0,dy=0;for(int c=0;c<channels;++c){int sx=0,sy=0;for(int j=-1;j<=1;++j)for(int i=-1;i<=1;++i){int v=src[(clamp15(y+j,h)+ry)*stride+(clamp15(x+i,w)+rx)*channels+c];sx+=v*i*(j==0?2:1);sy+=v*j*(i==0?2:1);}int norm=sx*sx+sy*sy;if(norm>best){best=norm;dx=sx;dy=sy;}}gx[t]=dx;gy[t]=dy;mag[t]=sqrt15(float(best));}}
__global__ void nms15(const int*gx,const int*gy,const float*mag,unsigned char*allowed,unsigned char*strong,int w,int h,int low,int high){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){int x=t%w,y=t/w,dx=gx[t],dy=gy[t],ax=dx<0?-dx:dx,ay=dy<0?-dy:dy;int ox=1,oy=0;if(float(ay)>2.41421356237f*ax){ox=0;oy=1;}else if(float(ay)>.41421356237f*ax)oy=(dx<0)==(dy<0)?1:-1;float a=mag[clamp15(y+oy,h)*w+clamp15(x+ox,w)],b=mag[clamp15(y-oy,h)*w+clamp15(x-ox,w)],v=mag[t];bool maximum=v>=a&&v>b;allowed[t]=maximum&&v>=low;strong[t]=maximum&&v>=high;}}
__device__ bool neighbor15(const unsigned char*src,int from,int to,int w,int stride,int rx,int ry,int mode){if(mode==1||mode==2){for(int c=0;c<3;++c){int v=int(src[(to/w+ry)*stride+(to%w+rx)*3+c])-int(src[(from/w+ry)*stride+(from%w+rx)*3+c]);if(v< -52||v>3)return false;}}return true;}
__global__ void seed15(const unsigned char*src,unsigned char*eligible,unsigned char*visited,int w,int h,int stride,int rx,int ry,int sx,int sy,int mode){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){bool yes=true;int low[3]={24,24,62},high[3]={240,248,251};for(int c=0;c<3;++c){auto v=src[(t/w+ry)*stride+(t%w+rx)*3+c];if(mode==0)yes&=v==src[(sy+ry)*stride+(sx+rx)*3+c];else if(mode>=3)yes&=v>=low[c]&&v<=high[c];}eligible[t]=yes;visited[t]=yes&&t==sy*w+sx;}}
__global__ void propagate15(const unsigned char*src,const unsigned char*allowed,const unsigned char*old,unsigned char*out,int w,int h,int stride,int rx,int ry,int mode,int connectivity,int*changed){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){bool yes=old[t];if(!yes&&allowed[t])for(int j=-1;j<=1&&!yes;++j)for(int i=-1;i<=1;++i){if((!i&&!j)||(connectivity==4&&i&&j))continue;int x=t%w+i,y=t/w+j;if(x>=0&&x<w&&y>=0&&y<h){int neighbor=y*w+x;if(old[neighbor]&&(mode<0||neighbor15(src,neighbor,t,w,stride,rx,ry,mode))){yes=true;break;}}}out[t]=yes;if(yes!=bool(old[t]))atomicExch(changed,1);}}
__global__ void paint15(const unsigned char*src,const unsigned char*visited,unsigned char*out,int w,int h,int stride,int rx,int ry,int mode,int channels,int outstride,int*stats){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){int x=t%w,y=t/w;bool boundary=false;if(visited[t]){atomicAdd(stats,1);atomicMin(stats+1,x);atomicMin(stats+2,y);atomicMax(stats+3,x);atomicMax(stats+4,y);for(int j=-1;j<=1;++j)for(int i=-1;i<=1;++i)if(x+i<0||x+i>=w||y+j<0||y+j>=h||!visited[(y+j)*w+x+i])boundary=true;if(boundary)atomicAdd(stats+5,1);}if(mode<0)out[y*outstride+x]=visited[t]?255:0;else{int colors[3]={255,0,0};if(mode==3){colors[0]=0;colors[1]=255;colors[2]=0;}if(mode==4){colors[0]=128;colors[1]=64;colors[2]=0;}if((mode==2||mode==4)&&boundary){colors[0]=128;colors[1]=0;colors[2]=255;}for(int c=0;c<channels;++c)out[y*outstride+x*channels+c]=visited[t]?colors[c]:src[(y+ry)*stride+(x+rx)*channels+c];}}}
__global__ void distance_row15(const unsigned char*src,int*distance,int*site,int w,int h,int stride,int rx,int ry){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){int x=t%w,y=t/w,best=100000000,at=-1;if(src[(y+ry)*stride+x+rx]==0){best=0;at=x;}else for(int j=0;j<w;++j)if(src[(y+ry)*stride+j+rx]==0){int d=(x-j)*(x-j);if(d<best){best=d;at=j;}}distance[t]=best;site[t]=at;}}
__global__ void distance_col15(const int*rows,const int*site,float*out,unsigned short*truncated,short*voronoi,int w,int h){int t=blockIdx.x*blockDim.x+threadIdx.x;if(t<w*h){int x=t%w,y=t/w,best=100000000,xx=-1,yy=-1;for(int j=0;j<h;++j){int d=rows[j*w+x]+(y-j)*(y-j);if(d<best){best=d;xx=site[j*w+x];yy=j;}}float distance=xx<0?INFINITY:sqrt15(float(best));out[t]=distance;truncated[t]=xx<0?65535:(unsigned short)distance;voronoi[t*2]=xx;voronoi[t*2+1]=yy;}}
__global__ void ncc15(const unsigned char*src,const unsigned char*tpl,float*out,int w,int h,int stride,int rx,int ry,int tw,int th){int t=blockIdx.x*blockDim.x+threadIdx.x;int ow=w-tw+1,oh=h-th+1;if(t<ow*oh){int x=t%ow,y=t/ow;long long sx=0,sy=0,sxx=0,syy=0,sxy=0;for(int j=0;j<th;++j)for(int i=0;i<tw;++i){int a=src[(y+j+ry)*stride+x+i+rx],b=tpl[i+j*tw];sx+=a;sy+=b;sxx+=a*a;syy+=b*b;sxy+=a*b;}double n=tw*th,aa=n*sxx-double(sx)*sx,bb=n*syy-double(sy)*sy;out[t]=(aa<=0||bb<=0)?0:float((n*sxy-double(sx)*sy)/root15(aa*bb));}}
__global__ void max15(const float*scores,int n,int*index){if(!threadIdx.x){int at=0;for(int i=1;i<n;++i)if(scores[i]>scores[at])at=i;*index=at;}}
