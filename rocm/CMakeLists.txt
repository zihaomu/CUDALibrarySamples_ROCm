cmake_minimum_required(VERSION 3.21)
project(cuda_library_samples_rocm LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../migration/environments/project.lock.json" _project_lock)
string(JSON _locked_arch GET "${_project_lock}" gpu_architecture)
set(MIGRATION_AMD_ARCH "${_locked_arch}" CACHE STRING "AMD architecture selected by the project profile")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# All current M1 programs call precompiled GPU libraries from C++ host code.
# Future custom device kernels will require separately qualified HIP targets.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} -c
  "import pathlib, importlib.metadata; import _rocm_sdk_core; assert importlib.metadata.version('rocm-sdk-core') == '10.1.0'; print(pathlib.Path(_rocm_sdk_core.__file__).parent)"
  OUTPUT_VARIABLE _sdk_core OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _sdk_status)
if(NOT _sdk_status EQUAL 0)
  message(FATAL_ERROR "Configure inside the locked ROCm 10.1.0 foundation")
endif()
get_filename_component(_sdk_site "${_sdk_core}" DIRECTORY)
set(ROCM_CORE_ROOT "${_sdk_core}" CACHE PATH "Locked ROCm SDK core")
set(ROCM_LIBRARIES_ROOT "${_sdk_site}/_rocm_sdk_libraries" CACHE PATH "Locked math runtime")
set(ROCM_MATH_INCLUDE "/opt/rocm/core-10.1/include" CACHE PATH "Matching 10.1 math headers")
file(GLOB _runtime "${ROCM_CORE_ROOT}/lib/libamdhip64.so*")
list(SORT _runtime)
list(GET _runtime 0 _runtime_lib)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../migration/m1.json" _specs)
string(JSON _count LENGTH "${_specs}")
math(EXPR _last "${_count}-1")
enable_testing()
foreach(_i RANGE ${_last})
  string(JSON _name GET "${_specs}" ${_i} name)
  string(JSON _path GET "${_specs}" ${_i} path)
  string(JSON _lib GET "${_specs}" ${_i} libraries 0)
  file(GLOB _candidates "${ROCM_LIBRARIES_ROOT}/lib/lib${_lib}.so*")
  if(NOT _candidates)
    message(FATAL_ERROR "Required target ${_name} is missing ${_lib} in the locked environment")
  endif()
  list(SORT _candidates)
  list(GET _candidates 0 _shared)
  add_executable(${_name} "${_path}/sample.cpp")
  target_compile_definitions(${_name} PRIVATE __HIP_PLATFORM_AMD__)
  target_include_directories(${_name} PRIVATE common "${ROCM_CORE_ROOT}/include" "${ROCM_MATH_INCLUDE}")
  target_link_libraries(${_name} PRIVATE "${_shared}" "${_runtime_lib}")
  set_target_properties(${_name} PROPERTIES BUILD_RPATH "${ROCM_LIBRARIES_ROOT}/lib;${ROCM_CORE_ROOT}/lib")
  add_test(NAME ${_name} COMMAND ${_name})
  set_tests_properties(${_name} PROPERTIES ENVIRONMENT "MIGRATION_FIXTURE=${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures/m1-input.f64" TIMEOUT 180)
endforeach()

# Keep the original FFT normalization on GPU; project-only HIP overlay is qualified.
find_program(MIGRATION_HIPCC hipcc REQUIRED)
set(_scale_obj "${CMAKE_CURRENT_BINARY_DIR}/migration_scaling.o")
set(_overlay "${CMAKE_CURRENT_SOURCE_DIR}/../build/rocm-sdk-overlay")
add_custom_command(OUTPUT "${_scale_obj}"
 COMMAND "${MIGRATION_HIPCC}" "--rocm-path=${_overlay}" "--hip-path=${_overlay}"
 "--rocm-device-lib-path=${ROCM_CORE_ROOT}/lib/llvm/amdgcn/bitcode"
 "--offload-arch=${MIGRATION_AMD_ARCH}" -c "${CMAKE_CURRENT_SOURCE_DIR}/common/scaling.hip" -o "${_scale_obj}"
 DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/common/scaling.hip" VERBATIM)
set_source_files_properties("${_scale_obj}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
foreach(_fft cuFFT_1d_c2c cuFFT_1d_r2c_c2r)
 if(TARGET ${_fft})
  target_sources(${_fft} PRIVATE "${_scale_obj}")
 endif()
endforeach()
